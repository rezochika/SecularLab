cmake_minimum_required(VERSION 3.21)
project(SecularLab VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SECULAR_ENABLE_SANITIZERS "Enable address/undefined sanitizers on GCC/Clang" OFF)
option(SECULAR_LONG_TESTS "Register slow oil-rig integration tests" OFF)
find_package(Threads REQUIRED)
find_path(GMP_INCLUDE_DIR gmp.h PATH_SUFFIXES x86_64-linux-gnu aarch64-linux-gnu)
find_path(GMPXX_INCLUDE_DIR gmpxx.h)
find_library(GMP_LIBRARY NAMES gmp libgmp)
find_library(GMPXX_LIBRARY NAMES gmpxx libgmpxx)
find_path(BOOST_INCLUDE_DIR boost/multiprecision/gmp.hpp)
if(NOT GMP_INCLUDE_DIR OR NOT GMPXX_INCLUDE_DIR OR NOT GMP_LIBRARY OR NOT GMPXX_LIBRARY OR NOT BOOST_INCLUDE_DIR)
  message(FATAL_ERROR "GMP (including gmpxx) and Boost headers required. See README.md for CLion/MSYS2/Linux/macOS instructions; set CMAKE_PREFIX_PATH to their installation prefix.")
endif()
add_library(secular_core src/exact.cpp src/solver.cpp)
target_include_directories(secular_core PUBLIC include)
target_include_directories(secular_core SYSTEM PUBLIC ${GMP_INCLUDE_DIR} ${GMPXX_INCLUDE_DIR} ${BOOST_INCLUDE_DIR})
target_link_libraries(secular_core PUBLIC ${GMPXX_LIBRARY} ${GMP_LIBRARY} Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(secular_core PRIVATE -Wall -Wextra -Wpedantic)
  if(SECULAR_ENABLE_SANITIZERS)
    target_compile_options(secular_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(secular_core PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(secular src/main.cpp)
target_link_libraries(secular PRIVATE secular_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(secular_tests tests/test_core.cpp)
  target_link_libraries(secular_tests PRIVATE secular_core)
  add_test(NAME exact_and_certified_regression COMMAND secular_tests)
  add_executable(secular_algebra_tests tests/test_algebra.cpp)
  target_link_libraries(secular_algebra_tests PRIVATE secular_core)
  add_test(NAME extended_exact_algebra COMMAND secular_algebra_tests)
  add_executable(secular_recovery_tests tests/test_recovery.cpp)
  target_link_libraries(secular_recovery_tests PRIVATE secular_core)
  add_test(NAME clustered_and_scaled_recovery COMMAND secular_recovery_tests)
  set_tests_properties(clustered_and_scaled_recovery PROPERTIES TIMEOUT 180)
  add_executable(secular_exact_fast_tests tests/test_exact_fast.cpp)
  target_link_libraries(secular_exact_fast_tests PRIVATE secular_core)
  add_test(NAME fast_exact_arithmetic COMMAND secular_exact_fast_tests)
  add_executable(secular_certification_tests tests/test_certification.cpp)
  target_link_libraries(secular_certification_tests PRIVATE secular_core)
  add_test(NAME polynomial_and_inertia_certification COMMAND secular_certification_tests)
  set_tests_properties(polynomial_and_inertia_certification PROPERTIES TIMEOUT 180)
  add_executable(secular_parallel_exact_tests tests/test_parallel_exact.cpp)
  target_link_libraries(secular_parallel_exact_tests PRIVATE secular_core)
  add_test(NAME parallel_recurrence_and_modular_oracle COMMAND secular_parallel_exact_tests)
  set_tests_properties(parallel_recurrence_and_modular_oracle PROPERTIES TIMEOUT 180)
  find_package(Python3 3.8 COMPONENTS Interpreter QUIET)
  if(Python3_Interpreter_FOUND)
    add_test(NAME cli_and_matrix_integration
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_cli.py
              $<TARGET_FILE:secular> ${CMAKE_CURRENT_SOURCE_DIR})
    set_tests_properties(cli_and_matrix_integration PROPERTIES TIMEOUT 180)
    add_test(NAME matrix_file_integrity
      COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/check_matrices.py)
    foreach(order IN ITEMS 8 16 32)
      add_test(NAME known_spectrum_${order}
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_known_spectrum.py
                $<TARGET_FILE:secular> --orders ${order})
      set_tests_properties(known_spectrum_${order} PROPERTIES
        TIMEOUT 180 LABELS "known-spectrum" PROCESSORS 4)
    endforeach()
    if(SECULAR_LONG_TESTS)
      add_test(NAME oilrig_66_full_certification
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_oilrig.py
                $<TARGET_FILE:secular> ${CMAKE_CURRENT_SOURCE_DIR})
      set_tests_properties(oilrig_66_full_certification PROPERTIES
        TIMEOUT 7200 LABELS "slow;benchmark" RUN_SERIAL TRUE)
    endif()
  else()
    message(STATUS "Python 3 not found: optional CLI/data tests are not registered.")
  endif()
endif()
